const grid = document.getElementById("scenarioGrid");

const search = document.getElementById("search");

let currentFilter = "all";

function renderScenarios() {

  const query = search.value.toLowerCase().trim();

  const results = scenarios.filter(scenario => {

    const roleMatch =

      currentFilter === "all" ||

      scenario.role === currentFilter;

    const text = [

      scenario.title,

      scenario.subtitle,

      scenario.tag,

      scenario.description,

      scenario.role

    ].join(" ").toLowerCase();

    return roleMatch && text.includes(query);

  });

  grid.innerHTML = results.length

    ? results.map(scenario => `

      <article class="card scenario-card">

        <div class="scenario-topline">

          <span class="role-badge">

            ${scenario.role.toUpperCase()}

          </span>

          <span class="tag">${scenario.tag}</span>

        </div>

        <h3>${scenario.title}</h3>

        <strong>${scenario.subtitle}</strong>

        <p>${scenario.description}</p>

        <a

          class="button"

          href="${scenario.url}"

          target="_blank"

          rel="noopener noreferrer"

        >

          Open scenario ↗

        </a>

      </article>

    `).join("")

    : `

      <article class="card">

        <h3>No scenarios found</h3>

        <p>Try another search term or role filter.</p>

      </article>

    `;

}

document.querySelectorAll(".filter").forEach(button => {

  button.addEventListener("click", () => {

    document.querySelectorAll(".filter")

      .forEach(item => item.classList.remove("active"));

    button.classList.add("active");

    currentFilter = button.dataset.filter;

    renderScenarios();

  });

});

search.addEventListener("input", renderScenarios);
